# [주의] 이 명령어는 로컬 PC에서 실행해야 하며,
#        터널링을 사용하는 동안 이 터미널을 닫으면 안 됩니다.
#        키 파일 경로("C:\Users\yhc65\.ssh\gcp_bastion")는 사용자별로 다를 수 있습니다.

ssh -i "C:\Users\yhc65\.ssh\gcp_bastion" -N -L 5432:10.0.1.2:5432 hochan@35.216.120.175

# --- 명령어 구성 요소 ---
# -i "[키 경로]": SSH 개인 키 경로
# -N: 원격 셸 명령 없이 순수 포트 포워딩만 수행
# -L [로컬포트]:[DB 내부 IP]:[DB 포트]: 로컬 5432 -> DB VM 10.0.1.2:5432 연결 설정
# hochan@35.216.120.175: 접속할 배스천 VM의 사용자 이름과 외부 IP
